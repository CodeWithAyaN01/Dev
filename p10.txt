import numpy as np
import matplotlib.pyplot as plt
import seaborn as sns
from sklearn.datasets import load_breast_cancer
from sklearn.cluster import KMeans
from sklearn.decomposition import PCA
from sklearn.preprocessing import StandardScaler

# 1. Load dataset
data = load_breast_cancer()
X = data.data 

# 2. Preprocessing: Standardize features (Mean=0, Variance=1)
scaler = StandardScaler()
X_scaled = scaler.fit_transform(X)

# 3. K-Means Clustering: Grouping into 2 clusters
kmeans = KMeans(n_clusters=2, random_state=42, n_init=10)
clusters = kmeans.fit_predict(X_scaled)

# 4. Dimensionality Reduction: PCA (Compress 30 features into 2 components)
pca = PCA(n_components=2)
X_pca = pca.fit_transform(X_scaled)

# 5. Transform Centroids: Move 30D cluster centers to 2D space
centroids_original = kmeans.cluster_centers_
centroids_pca = pca.transform(centroids_original) 

# 6. Visualization
plt.figure(figsize=(10, 7))
colors = ["#e74c3c", "#3498db"] # Red and Blue

for cluster, color in zip(range(2), colors):
    plt.scatter(
        X_pca[clusters == cluster, 0], 
        X_pca[clusters == cluster, 1],
        color=color, alpha=0.6, edgecolor="k", label=f"Cluster {cluster}"
    )

# Plot Centroids
plt.scatter(
    centroids_pca[:, 0], centroids_pca[:, 1],
    s=250, c='black', marker='X', label="Centroids"
)

plt.legend(loc="upper right")
plt.title("K-Means Clustering on Breast Cancer Dataset (PCA-Reduced)")
plt.xlabel("Principal Component 1")
plt.ylabel("Principal Component 2")
plt.grid(True, linestyle='--', alpha=0.5)
plt.show()
